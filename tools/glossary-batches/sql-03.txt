# カテゴリ || 用語 || 説明 || コード例
JSON || ->> (JSON 取り出し) || JSON から値を文字列として取り出す（PostgreSQL・MySQL）。 || SELECT data->>'name' FROM t;
JSON || -> (JSON 取り出し) || JSON から値を JSON のまま取り出す。 || SELECT data->'address' FROM t;
JSON || #>> (JSON パス) || 入れ子の JSON をパスで指定して文字列で取り出す（PostgreSQL）。 || SELECT data#>>'{address,city}' FROM t;
JSON || JSON_EXTRACT || JSON から指定パスの値を取り出す（MySQL）。 || SELECT JSON_EXTRACT(data, '$.name') FROM t;
JSON || JSON_VALUE || JSON からスカラ値を取り出す（SQL Server・Oracle）。 || SELECT JSON_VALUE(data, '$.name') FROM t;
JSON || JSON_QUERY || JSON から配列やオブジェクトを取り出す。 || SELECT JSON_QUERY(data, '$.items') FROM t;
JSON || JSON_OBJECT || キーと値から JSON オブジェクトを作る。 || SELECT JSON_OBJECT('id', id, 'name', name) FROM t;
JSON || JSON_ARRAY || 値から JSON 配列を作る。 || SELECT JSON_ARRAY(1, 2, 3);
JSON || JSON_AGG / JSON_ARRAYAGG || 複数行を1つの JSON 配列にまとめる。 || SELECT JSON_AGG(name) FROM t;
JSON || JSONB || 検索しやすい形で保存する JSON 型（PostgreSQL）。 || CREATE TABLE t (data JSONB);
JSON || @> (JSON 含む) || JSON が指定の内容を含むか調べる（PostgreSQL）。 || SELECT * FROM t WHERE data @> '{"role":"admin"}';
JSON || jsonb_set || JSON の一部の値を差し替える。 || UPDATE t SET data = jsonb_set(data, '{age}', '20');
JSON || JSON_TABLE || JSON を表の形に展開する。 || SELECT * FROM JSON_TABLE('[1,2]', '$[*]' COLUMNS (n INT PATH '$')) j;
JSON || jsonb_each || JSON のキーと値を行に展開する。 || SELECT * FROM jsonb_each('{"a":1,"b":2}');
JSON || jsonb_array_elements || JSON 配列を1要素ずつの行に展開する。 || SELECT jsonb_array_elements('[1,2,3]');
JSON || OPENJSON || JSON を行と列に展開する（SQL Server）。 || SELECT * FROM OPENJSON('[{"a":1}]') WITH (a INT);
JSON || FOR JSON || 検索結果を JSON として出力する（SQL Server）。 || SELECT id, name FROM t FOR JSON AUTO;
JSON || JSON_VALID || 文字列が正しい JSON か調べる。 || SELECT JSON_VALID('{"a":1}');
配列 || ARRAY 型 || 1つの列に複数の値を持てる型（PostgreSQL）。 || CREATE TABLE t (tags TEXT[]);
配列 || ARRAY[...] || 配列の値を作る。 || SELECT ARRAY[1, 2, 3];
配列 || ANY (配列) || 配列のどれかと等しいか調べる。 || SELECT * FROM t WHERE id = ANY(ARRAY[1, 2]);
配列 || unnest || 配列を1要素ずつの行に展開する。 || SELECT unnest(ARRAY['a', 'b']);
配列 || array_agg || 複数行を1つの配列にまとめる。 || SELECT array_agg(name) FROM t;
配列 || array_length || 配列の要素数を返す。 || SELECT array_length(ARRAY[1,2,3], 1);
配列 || array_append || 配列の末尾に要素を足す。 || SELECT array_append(ARRAY[1,2], 3);
配列 || array_to_string || 配列を区切り文字でつないだ文字列にする。 || SELECT array_to_string(ARRAY['a','b'], ',');
配列 || string_to_array || 文字列を区切って配列にする。 || SELECT string_to_array('a,b,c', ',');
配列 || @> (配列 含む) || 配列が別の配列の要素をすべて含むか調べる。 || SELECT ARRAY[1,2,3] @> ARRAY[1,2];
配列 || && (配列 重なり) || 2つの配列に共通の要素があるか調べる。 || SELECT ARRAY[1,2] && ARRAY[2,3];
全文検索 || to_tsvector || 文章を全文検索用の語に分解する（PostgreSQL）。 || SELECT to_tsvector('english', 'a fat cat');
全文検索 || to_tsquery || 全文検索の検索語を作る。 || SELECT to_tsquery('english', 'cat & fat');
全文検索 || @@ (全文一致) || 文章が検索語に一致するか調べる。 || SELECT * FROM t WHERE to_tsvector(body) @@ to_tsquery('cat');
全文検索 || MATCH AGAINST || 全文検索を行う（MySQL）。 || SELECT * FROM t WHERE MATCH(body) AGAINST ('cat');
全文検索 || CREATE FULLTEXT INDEX || 全文検索用のインデックスを作る。 || CREATE FULLTEXT INDEX ft ON t(body);
全文検索 || ts_rank || 全文検索の一致度を点数にする。 || SELECT ts_rank(to_tsvector(body), to_tsquery('cat')) FROM t;
全文検索 || pg_trgm (類似検索) || 文字の並びが似ているものを探す拡張機能。 || SELECT similarity('apple', 'appel');
日付関数 || INTERVAL 加減算 || 日付に期間を足し引きする。 || SELECT NOW() + INTERVAL '7 days';
日付関数 || DATE_PART || 日付から年・月・日などを取り出す。 || SELECT DATE_PART('year', CURRENT_DATE);
日付関数 || YEAR / MONTH / DAY || 日付から年・月・日を取り出す（MySQL・SQL Server）。 || SELECT YEAR(d), MONTH(d), DAY(d) FROM t;
日付関数 || DATE_SUB || 日付から期間を引く（MySQL）。 || SELECT DATE_SUB(CURDATE(), INTERVAL 1 MONTH);
日付関数 || DATEADD || 日付に期間を足す（SQL Server）。 || SELECT DATEADD(day, 7, GETDATE());
日付関数 || GETDATE || 現在の日時を返す（SQL Server）。 || SELECT GETDATE();
日付関数 || SYSDATE || 現在の日時を返す（Oracle）。 || SELECT SYSDATE FROM DUAL;
日付関数 || ADD_MONTHS || 月を足した日付を返す（Oracle）。 || SELECT ADD_MONTHS(SYSDATE, 3) FROM DUAL;
日付関数 || MONTHS_BETWEEN || 2つの日付の月数の差を返す（Oracle）。 || SELECT MONTHS_BETWEEN(d2, d1) FROM t;
日付関数 || STR_TO_DATE || 文字列を日付に変える（MySQL）。 || SELECT STR_TO_DATE('2026-11-09', '%Y-%m-%d');
日付関数 || TO_TIMESTAMP || 文字列を日時に変える。 || SELECT TO_TIMESTAMP('2026-11-09 10:00', 'YYYY-MM-DD HH24:MI');
日付関数 || AT TIME ZONE || 時間帯（タイムゾーン）を変換する。 || SELECT NOW() AT TIME ZONE 'Asia/Tokyo';
日付関数 || TIMESTAMPTZ || 時間帯つきの日時型（PostgreSQL）。 || CREATE TABLE t (created_at TIMESTAMPTZ DEFAULT NOW());
日付関数 || EOMONTH || 月末の日付を返す（SQL Server）。 || SELECT EOMONTH('2026-11-09');
日付関数 || DAYOFWEEK / DOW || 曜日を数字で返す。 || SELECT EXTRACT(DOW FROM CURRENT_DATE);
日付関数 || generate_series (日付) || 連続した日付の行を作る（PostgreSQL）。 || SELECT generate_series('2026-11-01'::date, '2026-11-07', '1 day');
文字列関数 || REGEXP_REPLACE || 正規表現で文字列を置き換える。 || SELECT REGEXP_REPLACE('a1b2', '[0-9]', '');
文字列関数 || REGEXP_SUBSTR / REGEXP_MATCHES || 正規表現に合う部分を取り出す。 || SELECT REGEXP_SUBSTR('abc123', '[0-9]+');
文字列関数 || REGEXP_LIKE || 正規表現に合うか判定する。 || SELECT * FROM t WHERE REGEXP_LIKE(name, '^A');
文字列関数 || LOCATE || 文字列の位置を探す（MySQL）。 || SELECT LOCATE('b', 'abc');
文字列関数 || STRPOS || 文字列の位置を探す（PostgreSQL）。 || SELECT STRPOS('abc', 'b');
文字列関数 || OVERLAY || 文字列の一部を置き換える。 || SELECT OVERLAY('abcdef' PLACING 'XX' FROM 3 FOR 2);
文字列関数 || LEFT / RIGHT で切り出し || 先頭や末尾から指定文字数を取り出す。 || SELECT LEFT('abcdef', 3);
文字列関数 || MD5 || 文字列のハッシュ値（MD5）を返す。 || SELECT MD5('abc');
文字列関数 || ENCODE / DECODE (文字列) || バイナリと文字列を変換する（PostgreSQL）。 || SELECT ENCODE('abc'::bytea, 'base64');
文字列関数 || QUOTE_IDENT / QUOTE_LITERAL || 識別子や文字列を安全に引用符で囲む。 || SELECT QUOTE_LITERAL('it''s');
文字列関数 || STRING_SPLIT || 文字列を区切って行に展開する（SQL Server）。 || SELECT value FROM STRING_SPLIT('a,b,c', ',');
文字列関数 || REGEXP_SPLIT_TO_TABLE || 正規表現で区切って行に展開する（PostgreSQL）。 || SELECT REGEXP_SPLIT_TO_TABLE('a1b2c', '[0-9]');
文字列関数 || 照合順序 (COLLATE) || 文字の並びや大小比較のルール。 || SELECT * FROM t ORDER BY name COLLATE "C";
文字列関数 || 文字コード (UTF-8) || 文字を数字で表す規則。日本語は UTF-8 が標準。 || CREATE DATABASE d ENCODING 'UTF8';
数値関数 || ROUND の桁指定 || 小数の桁を指定して丸める。マイナスで整数の桁も丸められる。 || SELECT ROUND(1234.567, -2);
数値関数 || TRUNCATE (関数) || 小数を切り捨てる（MySQL）。 || SELECT TRUNCATE(3.789, 1);
数値関数 || PI || 円周率を返す。 || SELECT PI();
数値関数 || SIN / COS / TAN || 三角関数。 || SELECT SIN(PI() / 2);
数値関数 || DIV (整数除算) || 割り算の商（整数部）だけを返す。 || SELECT 7 DIV 2;
数値関数 || width_bucket || 数値を等間隔の区間に振り分ける。 || SELECT width_bucket(55, 0, 100, 10);
数値関数 || setseed / RANDOM の再現 || 乱数の種を固定して同じ結果を再現する。 || SELECT setseed(0.5);
数値関数 || MONEY 型 || 金額用のデータ型。丸め誤差が出る場合は NUMERIC がよい。 || price MONEY
数値関数 || 浮動小数点の誤差 || FLOAT は 0.1 + 0.2 が 0.3 にならないことがある。金額は DECIMAL を使う。 || SELECT 0.1::float + 0.2::float;
集計 || GROUP BY の列番号 || SELECT の何番目の列かで GROUP BY を指定する。 || SELECT dept, COUNT(*) FROM emp GROUP BY 1;
集計 || GROUP BY の別名 || SELECT で付けた別名を GROUP BY で使える（DB による）。 || SELECT YEAR(d) y, COUNT(*) FROM t GROUP BY y;
集計 || 条件付き集計 (CASE) || CASE と SUM を組み合わせて条件ごとに数える。 || SELECT SUM(CASE WHEN sex = 'M' THEN 1 ELSE 0 END) FROM t;
集計 || ピボット (PIVOT) || 行を列に並べ替えて集計表にする。 || SELECT * FROM t PIVOT (SUM(v) FOR m IN ([1], [2]));
集計 || crosstab || ピボット集計を行う関数（PostgreSQL の tablefunc）。 || SELECT * FROM crosstab('SELECT ...') AS ct(a TEXT, b INT);
集計 || MEDIAN / PERCENTILE_CONT || 中央値や百分位数を求める。 || SELECT PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY v) FROM t;
集計 || MODE || 最も多く出てくる値（最頻値）を求める。 || SELECT MODE() WITHIN GROUP (ORDER BY v) FROM t;
集計 || CORR / COVAR_POP || 2つの列の相関係数・共分散を求める。 || SELECT CORR(a, b) FROM t;
集計 || BOOL_AND / BOOL_OR || 全部が真か、どれかが真かを集計する。 || SELECT BOOL_AND(active) FROM t;
集計 || ARRAY_AGG ORDER BY || 並び順を指定して配列にまとめる。 || SELECT ARRAY_AGG(name ORDER BY id) FROM t;
集計 || GROUPING 関数 || ROLLUP の小計行かどうかを判定する。 || SELECT dept, GROUPING(dept) FROM emp GROUP BY ROLLUP(dept);
集計 || 重複行の検出 || GROUP BY と HAVING で重複している値を探す。 || SELECT email, COUNT(*) FROM t GROUP BY email HAVING COUNT(*) > 1;
集計 || 重複行の削除 || ROW_NUMBER で順位を付け、2番目以降を削除する。 || DELETE FROM t WHERE id IN (SELECT id FROM (SELECT id, ROW_NUMBER() OVER (PARTITION BY email ORDER BY id) rn FROM t) x WHERE rn > 1);
ウィンドウ関数 || NTH_VALUE || ウィンドウ内の n 番目の値を取る。 || SELECT NTH_VALUE(v, 2) OVER (ORDER BY d) FROM t;
ウィンドウ関数 || PERCENT_RANK || 順位を 0〜1 の割合で返す。 || SELECT PERCENT_RANK() OVER (ORDER BY score) FROM t;
ウィンドウ関数 || CUME_DIST || 累積の分布（その値以下の割合）を返す。 || SELECT CUME_DIST() OVER (ORDER BY score) FROM t;
ウィンドウ関数 || RANGE BETWEEN || 値の範囲でウィンドウの幅を決める。 || SUM(v) OVER (ORDER BY d RANGE BETWEEN INTERVAL '7 days' PRECEDING AND CURRENT ROW)
ウィンドウ関数 || 名前付きウィンドウ (WINDOW) || ウィンドウの指定に名前を付けて使い回す。 || SELECT SUM(v) OVER w FROM t WINDOW w AS (PARTITION BY g ORDER BY d);
ウィンドウ関数 || 前日比 (LAG) || LAG で前の行の値を取り、差や割合を求める。 || SELECT v - LAG(v) OVER (ORDER BY d) FROM t;
ウィンドウ関数 || 連番の欠け探し || ROW_NUMBER や LAG で番号の飛びを見つける。 || SELECT id FROM (SELECT id, LAG(id) OVER (ORDER BY id) p FROM t) x WHERE id - p > 1;
ウィンドウ関数 || グループ内 上位N件 || ROW_NUMBER で順位を付け、上位だけ取り出す。 || SELECT * FROM (SELECT *, ROW_NUMBER() OVER (PARTITION BY g ORDER BY v DESC) rn FROM t) x WHERE rn <= 3;
ウィンドウ関数 || FRAME EXCLUDE || ウィンドウの範囲から自分の行などを除く。 || SUM(v) OVER (ORDER BY d ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING EXCLUDE CURRENT ROW)
