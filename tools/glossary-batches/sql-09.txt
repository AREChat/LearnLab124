# カテゴリ || 用語 || 説明 || コード例
関数 || COUNT(1) と COUNT(*) || どちらも行数を数える。違いはほぼない。 || SELECT COUNT(1) FROM t;
関数 || SUM と NULL の扱い || NULL は無視して合計する。 || SELECT SUM(v) FROM t;
関数 || COALESCE の連鎖 || 最初の NULL でない値を使う。 || COALESCE(a, b, c, 0)
関数 || NULLIF の使い道 || 0 の割り算を避ける。 || a / NULLIF(b, 0)
関数 || GREATEST / LEAST || 複数の値の最大・最小を返す。 || SELECT GREATEST(1, 5, 3);
関数 || LENGTH とバイト数 || 文字数とバイト数の違い。日本語は注意。 || SELECT OCTET_LENGTH('あ');
関数 || SUBSTRING の位置 || 開始位置は 1 から始まる。 || SUBSTRING('abcdef', 2, 3)
関数 || REPLACE の連続 || 複数の文字を順に置き換える。 || REPLACE(REPLACE(s, 'a', 'b'), 'c', 'd')
関数 || LPAD でゼロ埋め || 数字を0で埋めて桁をそろえる。 || LPAD('5', 3, '0')
関数 || CONCAT と NULL || CONCAT は NULL を空とみなすことが多い。 || CONCAT('a', NULL, 'b')
関数 || 日付から文字列 (TO_CHAR) || 日付を書式指定で文字列にする。 || TO_CHAR(d, 'YYYY"年"MM"月"')
関数 || 日付の差 (日数) || 引き算で日数が出る（PostgreSQL）。 || SELECT d2 - d1;
関数 || 月の加算 || 月を足す。 || d + INTERVAL '1 month'
関数 || 週の取得 || 何週目かを調べる。 || EXTRACT(WEEK FROM d)
関数 || 四半期の取得 || 何四半期かを調べる。 || EXTRACT(QUARTER FROM d)
関数 || 年度の計算 || 4月始まりの年度を求める。 || EXTRACT(YEAR FROM d - INTERVAL '3 months')
関数 || UNIX 時間 || 秒数に変換する。 || EXTRACT(EPOCH FROM now())
関数 || 現在時刻と トランザクション時刻 || NOW は開始時刻、clock_timestamp は実際の時刻。 || SELECT clock_timestamp();
関数 || 乱数と範囲 || 1〜10 の整数を作る。 || FLOOR(RANDOM() * 10) + 1
関数 || 小数点以下の桁数 || 数値を指定の桁で丸める。 || ROUND(v::numeric, 2)
関数 || 数値のキャスト || 文字列を数値にする。 || CAST('12.5' AS NUMERIC)
関数 || 文字列から日付 || 文字列を日付にする。 || TO_DATE('2026-11-09', 'YYYY-MM-DD')
関数 || BOOLEAN の変換 || 文字を真偽値にする。 || 't'::boolean
関数 || ASCII と文字コード || 文字と番号を変換する。 || ASCII('A')
関数 || MD5 と SHA || ハッシュを求める。 || SHA256('abc')
関数 || 配列の長さ || 配列の要素数を数える。 || CARDINALITY(arr)
関数 || 文字列の繰り返し || 同じ文字をくり返す。 || REPEAT('ab', 3)
関数 || 文字の置換 (TRANSLATE) || 1文字ずつ対応して置き換える。 || TRANSLATE('abc', 'abc', 'xyz')
関数 || 大文字小文字変換 || 大文字・小文字にそろえる。 || LOWER(email)
関数 || 空白の除去 || 前後や全体の空白を除く。 || TRIM(BOTH ' ' FROM s)
関数 || 全角空白の除去 || 全角スペースも取り除く。 || REGEXP_REPLACE(s, '[\s　]+', '', 'g')
関数 || 数字だけを取り出す || 数字以外を消す。 || REGEXP_REPLACE(s, '[^0-9]', '', 'g')
関数 || 文字列の分割と n 番目 || 区切って n 番目を取る。 || SPLIT_PART('a,b,c', ',', 2)
関数 || 文字列の部分一致の位置 || 見つかる位置を返す。 || POSITION('b' IN 'abc')
関数 || 日本語の並び順 || 照合順序で並び順が変わる。 || ORDER BY name COLLATE "ja_JP"
関数 || 全角半角の統一 || 比較前に表記ゆれをそろえる。 || NORMALIZE(s, NFKC)
関数 || 条件分岐の入れ子 || CASE を入れ子にして複雑な条件を表す。 || CASE WHEN a THEN CASE WHEN b THEN 1 END END
関数 || CASE の評価順 || 最初に当てはまった WHEN が使われる。 || CASE WHEN v > 80 THEN 'A' WHEN v > 60 THEN 'B' END
関数 || 型の優先順位 || 異なる型の演算では、優先度の高い型に合わされる。 || 1 + 1.5
関数 || 整数の割り算 || 整数同士の割り算は切り捨てられる DB がある。 || SELECT 7 / 2;
関数 || 小数の割り算 || 片方を小数にすると小数の結果になる。 || SELECT 7 / 2.0;
関数 || ビット演算 || ビットごとの AND/OR/XOR。 || SELECT 5 & 3;
関数 || 16進数・2進数 || 数を16進や2進で表す。 || SELECT TO_HEX(255);
関数 || 角度とラジアン || 角度をラジアンに変える。 || RADIANS(180)
関数 || 距離の計算 (簡易) || 2点間の直線距離を求める。 || SQRT(POWER(x2 - x1, 2) + POWER(y2 - y1, 2))
関数 || 階乗・組み合わせ || 数学的な計算を行う。 || FACTORIAL(5)
関数 || 平方根 || 平方根を求める。 || SQRT(16)
関数 || 絶対値 || 符号を取り除く。 || ABS(-5)
関数 || 符号 || プラス・マイナスを判定する。 || SIGN(-5)
関数 || 切り上げ・切り捨て || 小数を整数にそろえる。 || CEIL(2.1), FLOOR(2.9)
関数 || 余りの計算 || 割った余りを求める。 || MOD(7, 3)
関数 || べき乗 || 累乗を求める。 || POWER(2, 10)
関数 || 自然対数・常用対数 || 対数を求める。 || LN(10), LOG(100)
関数 || 指数関数 || e のべき乗。 || EXP(1)
関数 || 三角関数 || 三角関数の値。 || SIN(0), COS(0)
関数 || 統計関数 (STDDEV) || 標準偏差を求める。 || STDDEV(v)
関数 || 分散 (VARIANCE) || 分散を求める。 || VARIANCE(v)
関数 || 相関係数 (CORR) || 2つの値の関係の強さ。 || CORR(a, b)
関数 || 回帰 (REGR_SLOPE) || 回帰直線の傾き。 || REGR_SLOPE(y, x)
関数 || 最頻値 (MODE) || 最も多い値。 || MODE() WITHIN GROUP (ORDER BY v)
関数 || 件数の多い順 || 出現回数で並べる。 || GROUP BY v ORDER BY COUNT(*) DESC
トラブル対策 || 文字化けの原因 || 文字コードの設定が合っていない。 || ENCODING 'UTF8'
トラブル対策 || 全角・半角でヒットしない || 表記ゆれで検索に失敗する。 || NORMALIZE で統一
トラブル対策 || 前後の空白による不一致 || 見えない空白で比較が合わない。 || TRIM で除去
トラブル対策 || 日付の月日が入れ替わる || 日付形式の解釈ミス。 || ISO 形式 (YYYY-MM-DD) を使う
トラブル対策 || タイムゾーンのずれ || 日時が9時間ずれる。 || TIMESTAMPTZ を使う
トラブル対策 || 浮動小数点の誤差 || 0.1 + 0.2 が 0.3 にならない。 || NUMERIC を使う
トラブル対策 || NULL の計算結果 || NULL を含む計算は NULL になる。 || COALESCE で補う
トラブル対策 || 重複登録 || 同時に登録して重複する。 || UNIQUE 制約で防ぐ
トラブル対策 || ロック待ちで固まる || 他の処理が終わるのを待ち続ける。 || トランザクションを短くする
トラブル対策 || 遅いクエリの原因 || インデックスなし、全件取得、N+1 など。 || EXPLAIN で調べる
トラブル対策 || メモリ不足 || 大量の結果や並べ替えでメモリが足りない。 || LIMIT や インデックスで減らす
トラブル対策 || 接続が切れる || 長時間の処理でタイムアウトする。 || keepalive の設定
トラブル対策 || 外部キーで消せない || 子の行が残っているため削除できない。 || CASCADE か先に子を削除
トラブル対策 || 列を足すと遅い || 大きな表への列追加は時間がかかる。 || 既定値なしで追加する
トラブル対策 || 型を変えると失敗する || 変換できない値が残っている。 || USING で変換方法を指定
トラブル対策 || 権限エラー || 権限が足りない。 || GRANT を確認する
トラブル対策 || 検索結果が多すぎる || 条件が足りず、大量の行が返る。 || WHERE と LIMIT を付ける
トラブル対策 || UPDATE の WHERE 忘れ || 全行を更新してしまう事故。 || 先に SELECT で確認し、トランザクションで実行
トラブル対策 || DELETE の WHERE 忘れ || 全行を削除してしまう事故。 || BEGIN してから実行し、確認後 COMMIT
