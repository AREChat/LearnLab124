# SQL 第1回（基本構文・データ型・関数・結合・サブクエリ・ウィンドウ関数）
データ型 || INTEGER || 整数を入れる型。INT と書くこともある。 || CREATE TABLE t (n INTEGER);
データ型 || SMALLINT || 小さめの整数型。 || CREATE TABLE t (n SMALLINT);
データ型 || BIGINT || 大きな整数型。 || CREATE TABLE t (n BIGINT);
データ型 || DECIMAL || 桁数を決められる正確な小数型。お金の計算に向く。 || CREATE TABLE t (price DECIMAL(10,2));
データ型 || NUMERIC || DECIMAL とほぼ同じ、正確な小数型。 || CREATE TABLE t (rate NUMERIC(5,3));
データ型 || FLOAT || 近似値の小数型。誤差が出ることがある。 || CREATE TABLE t (x FLOAT);
データ型 || REAL || 単精度の小数型。 || CREATE TABLE t (x REAL);
データ型 || DOUBLE PRECISION || 倍精度の小数型。 || CREATE TABLE t (x DOUBLE PRECISION);
データ型 || CHAR || 長さが固定の文字列型。足りない分は空白で埋まる。 || CREATE TABLE t (code CHAR(3));
データ型 || VARCHAR || 長さが変わる文字列型。 || CREATE TABLE t (name VARCHAR(50));
データ型 || TEXT || 長い文字列を入れる型（PostgreSQL・MySQLなど）。 || CREATE TABLE t (memo TEXT);
データ型 || DATE || 日付（年月日）の型。 || CREATE TABLE t (d DATE);
データ型 || TIME || 時刻の型。 || CREATE TABLE t (at TIME);
データ型 || TIMESTAMP || 日付と時刻をあわせた型。 || CREATE TABLE t (created TIMESTAMP);
データ型 || BOOLEAN || 真（TRUE）か偽（FALSE）を入れる型。 || CREATE TABLE t (ok BOOLEAN);
データ型 || BLOB || 画像などのバイナリデータを入れる型。 || CREATE TABLE t (img BLOB);
データ型 || INTERVAL || 時間の長さ（3日間など）を表す型。 || SELECT CURRENT_DATE + INTERVAL '7 days';
データ型 || UUID || 重複しにくい識別用の文字列型。 || CREATE TABLE t (id UUID PRIMARY KEY);
データ型 || JSON || JSON形式のデータを入れる型。 || CREATE TABLE t (data JSON);
データ型 || ENUM || あらかじめ決めた値だけを入れる型（MySQLなど）。 || CREATE TABLE t (size ENUM('S','M','L'));
データ型 || AUTO_INCREMENT || 行を追加するたびに番号が自動で増える設定（MySQL）。 || CREATE TABLE t (id INT AUTO_INCREMENT PRIMARY KEY);
データ型 || SERIAL || 自動で増える連番の整数型（PostgreSQL）。 || CREATE TABLE t (id SERIAL PRIMARY KEY);
データ型 || IDENTITY || 自動で連番をふる列の設定。 || CREATE TABLE t (id INT GENERATED ALWAYS AS IDENTITY);
基本 || NULL || 「値がない・わからない」ことを表す特別な値。0や空文字とは別もの。 || SELECT * FROM t WHERE x IS NULL;
基本 || AS (別名) || 列やテーブルに別名をつける。 || SELECT price AS 価格 FROM items;
基本 || * (アスタリスク) || すべての列を指定する記号。 || SELECT * FROM items;
基本 || LIMIT || 取り出す行数の上限を決める（MySQL・PostgreSQLなど）。 || SELECT * FROM items LIMIT 10;
基本 || OFFSET || 先頭から何行とばすかを決める。 || SELECT * FROM items LIMIT 10 OFFSET 20;
基本 || TOP || 先頭から指定行数だけ取り出す（SQL Server）。 || SELECT TOP 5 * FROM items;
基本 || FETCH FIRST || 先頭から指定行数だけ取り出す標準SQLの書き方。 || SELECT * FROM items FETCH FIRST 5 ROWS ONLY;
基本 || ASC || 小さい順（昇順）に並べる。 || SELECT * FROM items ORDER BY price ASC;
基本 || DESC || 大きい順（降順）に並べる。 || SELECT * FROM items ORDER BY price DESC;
基本 || NULLS FIRST / NULLS LAST || 並べ替えで NULL を先に置くか後に置くかを決める。 || SELECT * FROM t ORDER BY x NULLS LAST;
基本 || コメント (--) || 行の終わりまでをメモにして、実行されなくする。 || SELECT 1; -- ここはメモ
基本 || コメント (/* */) || 複数行にわたるメモを書く。 || /* メモ */ SELECT 1;
基本 || セミコロン (;) || SQL文の終わりを示す記号。 || SELECT 1;
基本 || DUAL || 計算だけしたいときに使う仮のテーブル（Oracle）。 || SELECT 1+1 FROM DUAL;
基本 || EXISTS || サブクエリの結果が1行でもあれば真になる。 || SELECT * FROM a WHERE EXISTS (SELECT 1 FROM b WHERE b.id = a.id);
基本 || NOT EXISTS || サブクエリの結果が1行もなければ真になる。 || SELECT * FROM a WHERE NOT EXISTS (SELECT 1 FROM b WHERE b.id = a.id);
基本 || NOT IN || 候補の一覧にない値を選ぶ。NULL が混じると結果が空になる点に注意。 || SELECT * FROM a WHERE id NOT IN (1,2,3);
基本 || ESCAPE || LIKE で % や _ を普通の文字として扱うための目印を決める。 || SELECT * FROM t WHERE s LIKE '50\%' ESCAPE '\';
基本 || ワイルドカード (%) || LIKE で「0文字以上のどんな文字列でも」を表す。 || SELECT * FROM t WHERE name LIKE '田%';
基本 || ワイルドカード (_) || LIKE で「ちょうど1文字」を表す。 || SELECT * FROM t WHERE code LIKE 'A_1';
基本 || ILIKE || 大文字小文字を区別せず文字を探す LIKE（PostgreSQL）。 || SELECT * FROM t WHERE name ILIKE 'abc%';
基本 || SIMILAR TO || 正規表現ふうの書き方で文字を探す（PostgreSQL）。 || SELECT * FROM t WHERE s SIMILAR TO '(a|b)%';
基本 || REGEXP || 正規表現で文字を探す（MySQLなど）。 || SELECT * FROM t WHERE s REGEXP '^[0-9]+$';
基本 || 文字列連結 (||) || 文字列をつなげる演算子。 || SELECT first||last FROM t;
基本 || 比較演算子 (=) || 2つの値が等しいかを調べる。 || SELECT * FROM t WHERE x = 1;
基本 || 比較演算子 (<>) || 2つの値が等しくないかを調べる。!= と書ける製品もある。 || SELECT * FROM t WHERE x <> 1;
基本 || 比較演算子 (<, >, <=, >=) || 大小を比べる。 || SELECT * FROM t WHERE x >= 10;
基本 || 算術演算子 (+ - * /) || 足す・引く・かける・割る。 || SELECT price * qty FROM t;
基本 || 剰余 (%, MOD) || 割り算の余りを求める。 || SELECT MOD(10,3);
結合 || INNER JOIN || 両方のテーブルで条件が合う行だけを結合する。 || SELECT * FROM a INNER JOIN b ON a.id = b.a_id;
結合 || LEFT OUTER JOIN || 左のテーブルの行は全部残し、右に相手がなければ NULL にする。 || SELECT * FROM a LEFT OUTER JOIN b ON a.id = b.a_id;
結合 || RIGHT OUTER JOIN || 右のテーブルの行を全部残す結合。 || SELECT * FROM a RIGHT OUTER JOIN b ON a.id = b.a_id;
結合 || FULL OUTER JOIN || 両方のテーブルの行を全部残す結合。 || SELECT * FROM a FULL OUTER JOIN b ON a.id = b.a_id;
結合 || CROSS JOIN || 全部の組み合わせを作る結合（直積）。 || SELECT * FROM a CROSS JOIN b;
結合 || NATURAL JOIN || 同じ名前の列を自動でつないで結合する。 || SELECT * FROM a NATURAL JOIN b;
結合 || 自己結合 (SELF JOIN) || 同じテーブル同士を結合する。上司と部下の関係などに使う。 || SELECT e.name, m.name FROM emp e JOIN emp m ON e.mgr_id = m.id;
結合 || ON || 結合の条件を書く。 || SELECT * FROM a JOIN b ON a.id = b.a_id;
結合 || USING || 同じ名前の列を指定して結合する。 || SELECT * FROM a JOIN b USING (id);
結合 || 外部キーでの結合 || 外部キーの列同士をつないで、関連する行を取り出す。 || SELECT o.id, c.name FROM orders o JOIN customers c ON o.cid = c.id;
結合 || LATERAL || 前にある表の列を、あとのサブクエリで使えるようにする。 || SELECT * FROM a, LATERAL (SELECT * FROM b WHERE b.a_id = a.id LIMIT 1) x;
結合 || 結合の旧式書き方 || FROM に表をカンマで並べ、WHERE で結合条件を書く古い書き方。 || SELECT * FROM a, b WHERE a.id = b.a_id;
サブクエリ || サブクエリ || SQLの中に入れた、もう1つの SELECT 文。 || SELECT * FROM t WHERE x > (SELECT AVG(x) FROM t);
サブクエリ || スカラサブクエリ || 1行1列の値だけを返すサブクエリ。 || SELECT name, (SELECT MAX(price) FROM items) FROM t;
サブクエリ || 相関サブクエリ || 外側のクエリの値を使うサブクエリ。行ごとに評価される。 || SELECT * FROM a WHERE x > (SELECT AVG(x) FROM a a2 WHERE a2.g = a.g);
サブクエリ || インラインビュー || FROM の中に書くサブクエリ。仮のテーブルとして使う。 || SELECT * FROM (SELECT id, x FROM t) v WHERE x > 1;
サブクエリ || WITH (共通テーブル式) || 名前をつけた仮の結果を先に作っておく。見やすく書ける。 || WITH v AS (SELECT * FROM t) SELECT * FROM v;
サブクエリ || WITH RECURSIVE || 自分自身を呼び出す再帰的な共通テーブル式。階層データをたどる。 || WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<5) SELECT * FROM n;
サブクエリ || ANY / SOME || サブクエリの結果のどれか1つでも条件を満たせば真。 || SELECT * FROM t WHERE x > ANY (SELECT y FROM u);
サブクエリ || UNION ALL || 2つの結果を重複を消さずにつなげる。 || SELECT a FROM t1 UNION ALL SELECT a FROM t2;
サブクエリ || MINUS || 1つ目の結果から2つ目の結果を引く（Oracle）。EXCEPT と同じ働き。 || SELECT a FROM t1 MINUS SELECT a FROM t2;
集計 || GROUP BY ROLLUP || 小計と総計をまとめて出す。 || SELECT a, b, SUM(x) FROM t GROUP BY ROLLUP (a, b);
集計 || GROUP BY CUBE || すべての組み合わせの小計を出す。 || SELECT a, b, SUM(x) FROM t GROUP BY CUBE (a, b);
集計 || GROUPING SETS || 集計したい列の組み合わせを自分で指定する。 || SELECT a, b, SUM(x) FROM t GROUP BY GROUPING SETS ((a), (b));
集計 || COUNT(DISTINCT) || 重複をのぞいた件数を数える。 || SELECT COUNT(DISTINCT city) FROM t;
集計 || STDDEV || 標準偏差を求める。 || SELECT STDDEV(x) FROM t;
集計 || VARIANCE || 分散を求める。 || SELECT VARIANCE(x) FROM t;
集計 || STRING_AGG || グループ内の文字列を区切り文字でつなげる（PostgreSQL・SQL Server）。 || SELECT STRING_AGG(name, ',') FROM t;
集計 || GROUP_CONCAT || グループ内の値をつなげて1つの文字列にする（MySQL）。 || SELECT GROUP_CONCAT(name) FROM t;
集計 || LISTAGG || グループ内の値をつなげる（Oracle）。 || SELECT LISTAGG(name, ',') WITHIN GROUP (ORDER BY name) FROM t;
集計 || FILTER (WHERE) || 集計に使う行を条件で絞る。 || SELECT COUNT(*) FILTER (WHERE x > 0) FROM t;
集計 || HAVING と WHERE の違い || WHERE は集計前の行を、HAVING は集計後のグループを絞り込む。 || SELECT g, COUNT(*) FROM t WHERE x>0 GROUP BY g HAVING COUNT(*)>1;
ウィンドウ関数 || OVER || 行を減らさずに集計や順位づけをするための指定。 || SELECT x, SUM(x) OVER () FROM t;
ウィンドウ関数 || PARTITION BY || ウィンドウ関数でグループ分けする。 || SELECT g, x, SUM(x) OVER (PARTITION BY g) FROM t;
ウィンドウ関数 || ROW_NUMBER || 行に連番をふる。同じ値でも別の番号になる。 || SELECT ROW_NUMBER() OVER (ORDER BY x) FROM t;
ウィンドウ関数 || RANK || 順位をつける。同じ値は同順位で、次の順位がとぶ。 || SELECT RANK() OVER (ORDER BY score DESC) FROM t;
ウィンドウ関数 || DENSE_RANK || 順位をつける。同じ値は同順位で、次の順位はとばない。 || SELECT DENSE_RANK() OVER (ORDER BY score DESC) FROM t;
ウィンドウ関数 || NTILE || 行を指定した数のグループに等分する。 || SELECT NTILE(4) OVER (ORDER BY x) FROM t;
ウィンドウ関数 || LAG || 1つ前の行の値を取り出す。 || SELECT x, LAG(x) OVER (ORDER BY d) FROM t;
ウィンドウ関数 || LEAD || 1つ後ろの行の値を取り出す。 || SELECT x, LEAD(x) OVER (ORDER BY d) FROM t;
ウィンドウ関数 || FIRST_VALUE || ウィンドウの最初の行の値を取り出す。 || SELECT FIRST_VALUE(x) OVER (PARTITION BY g ORDER BY d) FROM t;
ウィンドウ関数 || LAST_VALUE || ウィンドウの最後の行の値を取り出す。 || SELECT LAST_VALUE(x) OVER (PARTITION BY g ORDER BY d ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) FROM t;
ウィンドウ関数 || 累計 (SUM OVER) || 並べた順に足し合わせていく。 || SELECT d, SUM(x) OVER (ORDER BY d) FROM t;
ウィンドウ関数 || 移動平均 || 前後の数行ぶんの平均を出す。 || SELECT AVG(x) OVER (ORDER BY d ROWS BETWEEN 2 PRECEDING AND CURRENT ROW) FROM t;
ウィンドウ関数 || ROWS BETWEEN || ウィンドウに含める行の範囲を決める。 || SUM(x) OVER (ORDER BY d ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING)
ウィンドウ関数 || UNBOUNDED PRECEDING || 範囲の始まりを「いちばん最初の行」にする。 || SUM(x) OVER (ORDER BY d ROWS UNBOUNDED PRECEDING)
ウィンドウ関数 || CURRENT ROW || 範囲の端を「いまの行」にする。 || SUM(x) OVER (ORDER BY d ROWS BETWEEN 1 PRECEDING AND CURRENT ROW)
文字列関数 || UPPER || 英字を大文字にする。 || SELECT UPPER('abc');
文字列関数 || LOWER || 英字を小文字にする。 || SELECT LOWER('ABC');
文字列関数 || INITCAP || 単語の先頭を大文字にする（PostgreSQL・Oracle）。 || SELECT INITCAP('hello world');
文字列関数 || LEFT || 文字列の左から指定文字数を取り出す。 || SELECT LEFT('abcdef', 3);
文字列関数 || RIGHT || 文字列の右から指定文字数を取り出す。 || SELECT RIGHT('abcdef', 3);
文字列関数 || POSITION || 文字列の中で、探す文字が何文字目にあるかを返す。 || SELECT POSITION('c' IN 'abcdef');
文字列関数 || INSTR || 文字列の中で、探す文字の位置を返す（Oracle・MySQL）。 || SELECT INSTR('abcdef', 'c');
文字列関数 || CHARINDEX || 探す文字の位置を返す（SQL Server）。 || SELECT CHARINDEX('c', 'abcdef');
文字列関数 || LPAD || 左側を指定の文字で埋めて長さをそろえる。 || SELECT LPAD('7', 3, '0');
文字列関数 || RPAD || 右側を指定の文字で埋めて長さをそろえる。 || SELECT RPAD('7', 3, '0');
文字列関数 || REVERSE || 文字列を逆順にする。 || SELECT REVERSE('abc');
文字列関数 || REPEAT || 文字列を指定回数くり返す。 || SELECT REPEAT('ab', 3);
文字列関数 || CONCAT_WS || 区切り文字を指定して文字列をつなげる。 || SELECT CONCAT_WS('-', 'a', 'b', 'c');
文字列関数 || SPLIT_PART || 区切り文字で分けて、指定した番目の部分を取り出す（PostgreSQL）。 || SELECT SPLIT_PART('a,b,c', ',', 2);
文字列関数 || TRANSLATE || 文字を1文字ずつ別の文字に置き換える。 || SELECT TRANSLATE('abc', 'ab', 'xy');
文字列関数 || ASCII || 文字の文字コード番号を返す。 || SELECT ASCII('A');
文字列関数 || CHR / CHAR || 文字コード番号から文字を返す。 || SELECT CHR(65);
文字列関数 || OCTET_LENGTH || 文字列のバイト数を返す。 || SELECT OCTET_LENGTH('あ');
文字列関数 || CHAR_LENGTH || 文字列の文字数を返す。 || SELECT CHAR_LENGTH('あいう');
文字列関数 || FORMAT || 値を指定した書式の文字列にする。 || SELECT FORMAT(1234.5, 'N2');
数値関数 || ABS || 絶対値を返す。 || SELECT ABS(-5);
数値関数 || CEIL / CEILING || 小数点以下を切り上げる。 || SELECT CEIL(1.2);
数値関数 || FLOOR || 小数点以下を切り捨てる。 || SELECT FLOOR(1.8);
数値関数 || MOD || 割り算の余りを返す。 || SELECT MOD(10, 3);
数値関数 || SQRT || 平方根を返す。 || SELECT SQRT(16);
数値関数 || SIGN || 数の符号（正は1、負は-1、0は0）を返す。 || SELECT SIGN(-3);
数値関数 || EXP || 自然対数の底 e の累乗を返す。 || SELECT EXP(1);
数値関数 || LN / LOG || 対数を返す。 || SELECT LN(10);
数値関数 || RANDOM / RAND || 0以上1未満の乱数を返す。 || SELECT RANDOM();
数値関数 || GREATEST || 複数の値のうち最大のものを返す。 || SELECT GREATEST(1, 5, 3);
数値関数 || LEAST || 複数の値のうち最小のものを返す。 || SELECT LEAST(1, 5, 3);
日付関数 || NOW || 現在の日付と時刻を返す。 || SELECT NOW();
日付関数 || EXTRACT || 日付から年・月・日などの部分を取り出す。 || SELECT EXTRACT(YEAR FROM CURRENT_DATE);
日付関数 || DATE_TRUNC || 日付を月初・年初などに切り捨てる（PostgreSQL）。 || SELECT DATE_TRUNC('month', NOW());
日付関数 || DATE_ADD || 日付に日数などを足す（MySQL）。 || SELECT DATE_ADD('2026-01-01', INTERVAL 7 DAY);
日付関数 || DATEDIFF || 2つの日付の差を返す。 || SELECT DATEDIFF('2026-01-10', '2026-01-01');
日付関数 || TO_CHAR || 日付や数値を書式つきの文字列にする（PostgreSQL・Oracle）。 || SELECT TO_CHAR(NOW(), 'YYYY-MM-DD');
日付関数 || TO_DATE || 文字列を日付に変換する。 || SELECT TO_DATE('2026-01-01', 'YYYY-MM-DD');
日付関数 || AGE || 2つの日付の間の年月日を返す（PostgreSQL）。 || SELECT AGE('2026-11-09', '2000-01-01');
日付関数 || LAST_DAY || その月の最終日を返す（Oracle・MySQL）。 || SELECT LAST_DAY('2026-02-10');
日付関数 || DATE_FORMAT || 日付を書式つきの文字列にする（MySQL）。 || SELECT DATE_FORMAT(NOW(), '%Y/%m/%d');
日付関数 || CURRENT_USER || いまのユーザー名を返す。 || SELECT CURRENT_USER;
変換・条件 || NULLIF || 2つの値が同じなら NULL を返す。0での割り算よけに使う。 || SELECT x / NULLIF(y, 0) FROM t;
変換・条件 || IFNULL / NVL / ISNULL || NULL のときに代わりの値を返す（製品ごとに名前が違う）。 || SELECT IFNULL(x, 0) FROM t;
変換・条件 || IIF || 条件が真か偽かで別の値を返す（SQL Server）。 || SELECT IIF(x > 0, '正', '負以下') FROM t;
変換・条件 || IF (関数) || 条件が真か偽かで別の値を返す（MySQL）。 || SELECT IF(x > 0, 'yes', 'no') FROM t;
変換・条件 || DECODE || 値を順に比べて対応する値を返す（Oracle）。 || SELECT DECODE(g, 1, '男', 2, '女', '不明') FROM t;
変換・条件 || TRY_CAST || 変換できないときはエラーでなく NULL を返す（SQL Server）。 || SELECT TRY_CAST('abc' AS INT);
変換・条件 || :: (型変換) || 値の型を変える短い書き方（PostgreSQL）。 || SELECT '123'::INTEGER;
